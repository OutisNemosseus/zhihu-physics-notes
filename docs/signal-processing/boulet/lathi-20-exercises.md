<script>
(function () {
  'use strict';
  const exam = document.getElementById('lathi-exam');
  if (!exam || exam.dataset.ready) return;
  const article = exam.closest('.md-typeset') || exam.parentElement;
  const headings = Array.from(article.querySelectorAll('h3[id^="exercise-"]'));
  if (!headings.length) return;
  const storageKey = 'lathi-20-exercises:current:v1';
  const questions = headings.map((heading, index) => {
    let chapter = heading.previousElementSibling;
    while (chapter && chapter.tagName !== 'H2') chapter = chapter.previousElementSibling;
    const chapterName = chapter ? chapter.textContent.replace(/¶/g, '').trim() : '';
    const card = document.createElement('section');
    card.className = 'lathi-question';
    card.setAttribute('aria-labelledby', heading.id);
    card.dataset.exercise = heading.id;
    heading.before(card);
    let node = heading;
    while (node && (node === heading || !/^H[23]$/.test(node.tagName))) {
      const next = node.nextElementSibling;
      card.appendChild(node);
      node = next;
    }
    heading.tabIndex = -1;
    const label = document.createElement('p');
    label.className = 'lathi-chapter-label';
    label.textContent = chapterName;
    card.prepend(label);
    return { card, heading, title: heading.textContent.replace(/¶/g, '').trim(), index };
  });
  article.querySelectorAll('h2').forEach(heading => {
    if (/^第 [1-4] 章/.test(heading.textContent)) heading.classList.add('lathi-chapter-heading');
  });
  const picker = exam.querySelector('.lathi-question-picker');
  questions.forEach(question => {
    const button = document.createElement('button');
    button.type = 'button';
    button.textContent = String(question.index + 1).padStart(2, '0');
    button.title = question.title;
    button.setAttribute('aria-label', question.title);
    button.dataset.questionIndex = String(question.index);
    button.addEventListener('click', () => show(question.index, true));
    picker.appendChild(button);
  });
  const bottom = document.createElement('div');
  bottom.className = 'lathi-bottom-nav';
  bottom.appendChild(exam.querySelector('.lathi-toolbar').cloneNode(true));
  questions[questions.length - 1].card.after(bottom);
  const controls = [exam, bottom];
  let current = 0;
  function hashIndex() {
    let id;
    try { id = decodeURIComponent(location.hash.slice(1)); } catch (_) { return -1; }
    return questions.findIndex(question => question.heading.id === id ||
      (id && question.card.contains(document.getElementById(id))));
  }
  function show(index, navigate) {
    if (!Number.isInteger(index) || index < 0 || index >= questions.length) return;
    current = index;
    questions.forEach(question => { question.card.hidden = question.index !== current; });
    controls.forEach(control => {
      control.querySelector('[data-exam-prev]').disabled = current === 0;
      control.querySelector('[data-exam-next]').disabled = current === questions.length - 1;
      control.querySelector('.lathi-progress').textContent = '第 ' + (current + 1) + ' / ' + questions.length + ' 题';
    });
    picker.querySelectorAll('button').forEach((button, i) => {
      button.setAttribute('aria-current', i === current ? 'step' : 'false');
    });
    exam.querySelector('progress').value = current + 1;
    try { localStorage.setItem(storageKey, questions[current].heading.id); } catch (_) {}
    if (navigate) {
      const hash = '#' + questions[current].heading.id;
      if (location.hash !== hash) history.pushState(null, '', hash);
      questions[current].heading.focus({ preventScroll: true });
      exam.scrollIntoView({ behavior: 'auto', block: 'start' });
    }
  }
  controls.forEach(control => {
    control.querySelector('[data-exam-prev]').addEventListener('click', () => show(current - 1, true));
    control.querySelector('[data-exam-next]').addEventListener('click', () => show(current + 1, true));
  });
  article.addEventListener('keydown', event => {
    if (event.altKey || event.ctrlKey || event.metaKey || event.shiftKey ||
        event.target.closest('input, textarea, select, [contenteditable="true"]')) return;
    if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
      event.preventDefault();
      show(current + (event.key === 'ArrowRight' ? 1 : -1), true);
    }
  });
  function followHash() { const index = hashIndex(); if (index >= 0) show(index, false); }
  window.addEventListener('hashchange', followHash);
  window.addEventListener('popstate', followHash);
  let initial = hashIndex();
  if (initial < 0 && !location.hash) {
    try { initial = questions.findIndex(question => question.heading.id === localStorage.getItem(storageKey)); } catch (_) {}
  }
  exam.dataset.ready = 'true';
  article.classList.add('lathi-exam-ready');
  show(initial >= 0 ? initial : 0, false);
  exam.hidden = false;
})();

</script>

import numpy as np

def pasm(n):
    P = np.ones((n, n), dtype=int)
    for i in range(1, n):
        for j in range(1, n):
            P[i, j] = P[i - 1, j] + P[i, j - 1]
    return P
print(pasm(4))
print(pasm(7))

def sumdig(num):
    if num.__class__ is not int:
        raise TypeError
    if num < 0:
        num = -num
    if num < 10:
        return num
    return num % 10 + sumdig(num // 10)
print(sumdig(123456))
